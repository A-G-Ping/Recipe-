    function playTimerBell() {
      try {
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = "sine";
        osc.frequency.setValueAtTime(587.33, audioCtx.currentTime); // D5
        osc.frequency.exponentialRampToValueAtTime(880, audioCtx.currentTime + 0.1); // A5
        gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 1.2);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + 1.2);
      } catch (e) {
        console.log("Audio alert playback", e);
      }
    }

    // PARSE COOK TIME IN MINUTES FOR RELIABLE SORTING
    function parseDurationMinutes(timeStr) {
      if (!timeStr) return 9999;
      const s = String(timeStr).toLowerCase();
      if (s.includes("reference")) return 9999;
      let total = 0;
      let matched = false;
      const hrMatch = s.match(/(\d+(?:\.\d+)?)\s*(?:hr|hour)/);
      if (hrMatch) {
        total += parseFloat(hrMatch[1]) * 60;
        matched = true;
      }
      const minMatch = s.match(/(\d+(?:\.\d+)?)\s*(?:m|min)/);
      if (minMatch) {
        total += parseFloat(minMatch[1]);
        matched = true;
      }
      const secMatch = s.match(/(\d+(?:\.\d+)?)\s*(?:sec)/);
      if (secMatch) {
        total += parseFloat(secMatch[1]) / 60;
        matched = true;
      }
      if (!matched) {
        const anyNum = s.match(/(\d+(?:\.\d+)?)/);
        if (anyNum) return parseFloat(anyNum[1]);
        return 9999;
      }
      return total;
    }

    // RENDER RECIPE CARDS
    function renderRecipeCards() {
      const grid = document.getElementById("recipe-cards-grid");
      const emptyState = document.getElementById("empty-state");
      const countDisplay = document.getElementById("recipes-count-display");
      const catLabel = document.getElementById("active-category-label");

      if (!showFavoritesOnly) {
        catLabel.textContent = currentCategory;
      }

      // Filter
      let list = RECIPES_DATA.filter(recipe => {
        if (currentCategory !== "All Recipes" && recipe.category !== currentCategory) {
          return false;
        }
        if (showFavoritesOnly && !favorites.includes(recipe.id)) {
          return false;
        }
        if (searchQuery) {
          const matchTitle = recipe.title.toLowerCase().includes(searchQuery);
          const matchAuthor = recipe.author.toLowerCase().includes(searchQuery);
          const matchBadge = recipe.badge.toLowerCase().includes(searchQuery);
          const matchCategory = recipe.category.toLowerCase().includes(searchQuery);
          const matchNote = recipe.note && recipe.note.toLowerCase().includes(searchQuery);
          const matchIngredients = recipe.ingredients.some(ing => ing.item.toLowerCase().includes(searchQuery));
          const matchDirections = recipe.directions.some(d => d.toLowerCase().includes(searchQuery));
          const matchTags = recipe.tags && recipe.tags.some(t => t.toLowerCase().includes(searchQuery));
          return matchTitle || matchAuthor || matchBadge || matchCategory || matchNote || matchIngredients || matchDirections || matchTags;
        }
        return true;
      });

      // Sort
      if (sortOrder === "title-asc") {
        list.sort((a, b) => a.title.localeCompare(b.title));
      } else if (sortOrder === "title-desc") {
        list.sort((a, b) => b.title.localeCompare(a.title));
      } else if (sortOrder === "quickest") {
        list.sort((a, b) => parseDurationMinutes(a.time) - parseDurationMinutes(b.time));
      } else {
        list.sort((a, b) => a.page - b.page);
      }

      countDisplay.textContent = `${list.length} recipe${list.length === 1 ? '' : 's'}`;

      if (list.length === 0) {
        grid.innerHTML = "";
        emptyState.classList.remove("hidden");
        return;
      } else {
        emptyState.classList.add("hidden");
      }

      grid.innerHTML = list.map(recipe => {
        const isFav = favorites.includes(recipe.id);
        const cleanMixIns = recipe.ingredients
          .map(i => i.item.replace(/^\[[^\]]+\]\s*/, '').split(',')[0])
          .slice(0, 4)
          .join(', ');
        const suffix = recipe.ingredients.length > 4 ? '...' : '';

        return `
          <div class="recipe-paper paper-edge-shadow rounded-2xl border border-parchment-300 p-5 flex flex-col justify-between hover:border-terracotta-400 hover:-translate-y-1 transition duration-200 cursor-pointer relative group" onclick="openRecipeModal('${recipe.id}')">
            <div>
              <div class="flex items-start justify-between gap-2 mb-2">
                <span class="text-[10px] font-bold uppercase tracking-wider bg-parchment-200 text-parchment-800 px-2 py-0.5 rounded-md">
                  ${recipe.badge}
                </span>
                <span class="text-xs font-serif italic text-parchment-800/60">p. ${recipe.page}</span>
              </div>

              <h3 class="text-xl font-serif font-bold text-parchment-900 leading-snug group-hover:text-terracotta-600 transition-colors">
                ${recipe.title}
              </h3>
              <p class="text-xs font-serif italic text-amberwood-700 mt-0.5">${recipe.author}</p>

              <!-- Ingredients preview -->
              <div class="mt-4 pt-3 border-t border-dashed border-parchment-200 text-xs text-parchment-800/80 line-clamp-2">
                <span class="font-semibold text-parchment-900">Key Mix-ins:</span>
                ${cleanMixIns}${suffix}
              </div>
            </div>

            <div class="mt-4 pt-3 border-t border-parchment-200 flex items-center justify-between text-xs text-parchment-800">
              <div class="flex items-center space-x-3">
                <span class="flex items-center gap-1">
                  <i data-lucide="clock" class="w-3.5 h-3.5 text-terracotta-500"></i>
                  <span>${recipe.time}</span>
                </span>
                <span class="flex items-center gap-1">
                  <i data-lucide="flame" class="w-3.5 h-3.5 text-terracotta-500"></i>
                  <span>${recipe.oven}</span>
                </span>
              </div>
              <button
                onclick="event.stopPropagation(); toggleFavorite('${recipe.id}')"
                class="p-1.5 rounded-full hover:bg-parchment-200 text-parchment-400 hover:text-terracotta-500 transition"
                aria-label="${isFav ? 'Remove from favorites' : 'Save to heirloom favorites'}"
                title="${isFav ? 'Remove from favorites' : 'Save to heirloom favorites'}"
              >
                <i data-lucide="heart" class="w-4 h-4 ${isFav ? 'fill-terracotta-500 text-terracotta-500' : ''}"></i>
              </button>
            </div>
          </div>
        `;
      }).join("");

      safeCreateIcons();
    }

    function updateScaleButtons() {
      document.querySelectorAll(".scale-btn").forEach(btn => {

    function parseFractionToken(str) {
      const s = String(str).trim();
      const mixed = s.match(/^(\d+)\s+(\d+)\/(\d+)$/);
      if (mixed) return parseInt(mixed[1], 10) + parseInt(mixed[2], 10) / parseInt(mixed[3], 10);
      const frac = s.match(/^(\d+)\/(\d+)$/);
      if (frac) return parseInt(frac[1], 10) / parseInt(frac[2], 10);
      const num = parseFloat(s);
      return isNaN(num) ? null : num;
    }

    function formatSingleNumber(val) {
      if (val === null || isNaN(val) || val <= 0) return "";
      const whole = Math.floor(val + 1e-4);
      const rem = val - whole;
      const fractions = [
        [0.125, "⅛"],
        [0.25, "¼"],
        [0.3333, "⅓"],
        [0.375, "⅜"],
        [0.5, "½"],
        [0.625, "⅝"],
        [0.6667, "⅔"],
        [0.75, "¾"],
        [0.875, "⅞"]
      ];
      for (const [fVal, fChar] of fractions) {
        if (Math.abs(rem - fVal) < 0.04) {
          return whole > 0 ? `${whole} ${fChar}` : fChar;
        }
      }
      if (Math.abs(rem) < 0.04) return String(whole);
      if (Math.abs(rem - 1) < 0.04) return String(whole + 1);
      return val.toFixed(2).replace(/\.?0+$/, '');
    }

    function formatIngredientQty(qty, multiplier) {
      if (qty === null || qty === undefined || qty === "") return "";
      if (typeof qty === "number") {
        return formatSingleNumber(qty * multiplier);
      }
      const s = String(qty).trim();
      // Handle dual batch like "3 / 6" or "1/2 / 1"
      if (s.includes(" / ")) {
        const parts = s.split(" / ").map(p => {
          const v = parseFractionToken(p);
          return v !== null ? formatSingleNumber(v * multiplier) : p;
        });
        return parts.join(" / ");
      }
      // Handle range like "10-12" or "1 to 1 1/2" (skip scaling on Turkey Weight Reference p.105)
      if (activeRecipe && activeRecipe.page === 105) {
        return s;
      }
      const rangeMatch = s.match(/^([\d\s/]+?)(\s*(?:-|to)\s*)([\d\s/]+)$/i);
      if (rangeMatch) {
        const v1 = parseFractionToken(rangeMatch[1]);
        const sep = rangeMatch[2];
        const v2 = parseFractionToken(rangeMatch[3]);
        if (v1 !== null && v2 !== null) {
          return `${formatSingleNumber(v1 * multiplier)}${sep}${formatSingleNumber(v2 * multiplier)}`;
        }
      }
      // Handle "1+"
      const plusMatch = s.match(/^(\d+(?:\.\d+)?)\+$/);
      if (plusMatch) {
        return `${formatSingleNumber(parseFloat(plusMatch[1]) * multiplier)}+`;
      }
      const single = parseFractionToken(s);
      if (single !== null) {
        return formatSingleNumber(single * multiplier);
      }
      return s;
    }

    function renderModalIngredients() {
      const container = document.getElementById("modal-ingredients-list");
      if (!activeRecipe) return;

      container.innerHTML = activeRecipe.ingredients.map((ing, idx) => {
        const isChecked = checkedIngredients.has(idx);
        const scaledQty = formatIngredientQty(ing.qty, currentServingScale);
        const unitPart = ing.unit ? `${ing.unit}` : "";
        const quantityLabel = [scaledQty, unitPart].filter(Boolean).join(" ");

        return `
          <label for="ingredient-checkbox-${idx}" class="flex items-start space-x-2.5 p-2 rounded-lg hover:bg-parchment-100 cursor-pointer transition select-none ${isChecked ? 'bg-parchment-100/60' : ''}">
            <input
              type="checkbox"
              id="ingredient-checkbox-${idx}"
              ${isChecked ? 'checked' : ''}
              onchange="toggleIngredientChecked(${idx})"
              class="mt-1 w-4 h-4 text-terracotta-600 rounded border-parchment-300 focus:ring-terracotta-500 cursor-pointer"
            >
            <span class="leading-relaxed ${isChecked ? 'checked-ingredient' : 'text-parchment-900'}">
              ${quantityLabel ? `<strong class="font-semibold text-terracotta-700">${quantityLabel}</strong> ` : ''}
              <span>${ing.item}</span>
            </span>
          </label>
        `;
      }).join("");
    }

    function toggleIngredientChecked(idx) {
      if (checkedIngredients.has(idx)) {
        checkedIngredients.delete(idx);
      } else {
        checkedIngredients.add(idx);
      }
      renderModalIngredients();
    }

    function toggleStepCompleted(idx) {
      if (completedSteps.has(idx)) {
        completedSteps.delete(idx);
      } else {
        completedSteps.add(idx);
      }
      renderModalDirections();
    }

    function renderModalDirections() {
      const container = document.getElementById("modal-directions-list");
      if (!activeRecipe) return;

      container.innerHTML = activeRecipe.directions.map((step, idx) => {
        const isDone = completedSteps.has(idx);
        // Highlight bold step lead-in if formatted as "Lead-In: instruction"
        let formattedStep = step;
        const colonIdx = step.indexOf(":");
        if (colonIdx > 0 && colonIdx < 32) {
          const lead = step.slice(0, colonIdx);
          const rest = step.slice(colonIdx + 1);
          formattedStep = `<strong class="font-semibold text-parchment-900">${lead}:</strong>${rest}`;
        }

        return `
          <div onclick="toggleStepCompleted(${idx})" class="flex items-start space-x-3 p-3 rounded-xl border ${isDone ? 'bg-parchment-100/70 border-parchment-200 opacity-65' : 'bg-white/70 border-parchment-200/80 hover:bg-parchment-100/80'} cursor-pointer transition select-none">
            <span class="flex-shrink-0 w-6 h-6 rounded-full ${isDone ? 'bg-terracotta-600 text-white' : 'bg-parchment-200 text-terracotta-700'} font-bold text-xs flex items-center justify-center font-serif mt-0.5 shadow-sm">
              ${isDone ? '✓' : idx + 1}
            </span>
            <p class="leading-relaxed ${isDone ? 'line-through text-parchment-700' : 'text-parchment-900'}">${formattedStep}</p>
          </div>
        `;
      }).join("");
    }

    function toggleTimerDrawer() {
      timerDrawerOpen = !timerDrawerOpen;
      const drawer = document.getElementById("timer-drawer");
      if (timerDrawerOpen) {
        drawer.classList.remove("translate-y-96", "opacity-0", "pointer-events-none");
      } else {
        drawer.classList.add("translate-y-96", "opacity-0", "pointer-events-none");
      }
    }

    function setTimerDuration(minutes, label = `${minutes} Minute Timer`) {
      clearInterval(timerInterval);
      isTimerRunning = false;
      timerDurationSeconds = minutes * 60;
      timerRemainingSeconds = timerDurationSeconds;
      document.getElementById("timer-label").textContent = label;
      document.getElementById("timer-status").textContent = "Paused";
      document.getElementById("timer-toggle-btn").textContent = "Start";
      document.getElementById("timer-running-dot").classList.add("hidden");
      updateTimerDisplay();
      if (!timerDrawerOpen) toggleTimerDrawer();
    }

    function startQuickTimer(minutes, label) {
      setTimerDuration(minutes, label);
      toggleTimerState();
    }

    function toggleTimerState() {
      if (timerRemainingSeconds <= 0) {
        setTimerDuration(5);
      }

      if (isTimerRunning) {
        clearInterval(timerInterval);
        isTimerRunning = false;
        document.getElementById("timer-toggle-btn").textContent = "Resume";
        document.getElementById("timer-status").textContent = "Paused";
        document.getElementById("timer-running-dot").classList.add("hidden");
      } else {
        isTimerRunning = true;
        document.getElementById("timer-toggle-btn").textContent = "Pause";
        document.getElementById("timer-status").textContent = "Running";
        document.getElementById("timer-running-dot").classList.remove("hidden");

        timerInterval = setInterval(() => {
          timerRemainingSeconds--;
          updateTimerDisplay();

          if (timerRemainingSeconds <= 0) {
            clearInterval(timerInterval);
            isTimerRunning = false;
            document.getElementById("timer-toggle-btn").textContent = "Restart";
            document.getElementById("timer-status").textContent = "Time is Up!";
            document.getElementById("timer-running-dot").classList.add("hidden");
            playTimerBell();
            if (!timerDrawerOpen) toggleTimerDrawer();
          }
        }, 1000);
      }
    }

    function resetTimer() {
      clearInterval(timerInterval);
      isTimerRunning = false;
      timerRemainingSeconds = timerDurationSeconds;
      document.getElementById("timer-toggle-btn").textContent = "Start";
      document.getElementById("timer-status").textContent = "Reset";
      document.getElementById("timer-running-dot").classList.add("hidden");
      updateTimerDisplay();
    }

    function updateTimerDisplay() {
      const minutes = Math.floor(timerRemainingSeconds / 60);
      const seconds = timerRemainingSeconds % 60;
      const formatted = `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
      document.getElementById("timer-countdown").textContent = formatted;
      document.getElementById("timer-badge-text").textContent = isTimerRunning ? formatted : "Timer";
    }

    // KEYBOARD SHORTCUT: Esc to close modal
    document.addEventListener("keydown", (e) => {
      if (e.key === "Escape") {
        closeRecipeModal();
        if (timerDrawerOpen) toggleTimerDrawer();
      }
    });
